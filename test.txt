{{- if .Values.route.enabled }}
apiVersion: route.openshift.io/v1
kind: Route

metadata:
  name: {{ .Values.name }}-route
  labels:
    app: {{ .Values.name }}
    router: internal
  annotations:
    haproxy.router.openshift.io/timeout: 300s

spec:
  host: {{ .Values.urlName }}-{{ .Values.namespace }}.{{ .Values.gatewayHostname }}
  path: /

  to:
    kind: Service
    name: {{ .Values.name }}
    weight: 100

  port:
    targetPort: http

  tls:
    termination: edge
    insecureEdgeTerminationPolicy: Redirect

  wildcardPolicy: None
{{- end }}
